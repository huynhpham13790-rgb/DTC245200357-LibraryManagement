Luồng hoạt động chính:
\begin{enumerate}
    \item Người dùng truy cập \url{http://localhost}, Nginx trả mã 301 chuyển sang \url{https://localhost}.
    \item Nginx kết thúc TLS, gắn security header và chuyển tiếp yêu cầu tới dịch vụ tương ứng theo đường dẫn.
    \item Web App xử lý nghiệp vụ, truy vấn PostgreSQL qua mạng \texttt{backend\_net}.
    \item Prometheus định kỳ 5 giây thu thập metrics từ Web App và các exporter; Promtail đọc log Nginx và đẩy về Loki.
    \item Quản trị viên xem dashboard và truy vấn log trên Grafana tại \url{https://localhost/grafana/}.
\end{enumerate}

\begin{itemize}
    \item \textbf{books:} mỗi dòng là một đầu sách; \texttt{quantity} là tổng số bản. Số bản còn lại không lưu cố định mà được tính bằng \texttt{quantity} trừ số phiếu đang mượn, tránh sai lệch dữ liệu.
    \item \textbf{readers:} thông tin độc giả, mã thẻ là duy nhất, có hạn thẻ \texttt{card\_expiry}.
    \item \textbf{borrow\_records:} mỗi dòng là một phiếu mượn một cuốn sách; \texttt{status} nhận giá trị \texttt{borrowing} hoặc \texttt{returned}; phiếu quá hạn là phiếu \texttt{borrowing} có \texttt{due\_date} nhỏ hơn thời điểm hiện tại.
    \item \textbf{librarians:} tài khoản đăng nhập, mật khẩu được băm bằng thuật toán scrypt kèm salt ngẫu nhiên.
\end{itemize}
Lược đồ được ứng dụng tự khởi tạo và nâng cấp khi khởi động (\texttt{CREATE TABLE IF NOT EXISTS}, \texttt{ALTER TABLE \ldots{} ADD COLUMN IF NOT EXISTS}) nên dữ liệu cũ được giữ nguyên khi cập nhật phiên bản.

\section{Thiết kế xử lý mượn sách}
Nghiệp vụ mượn sách cần đảm bảo không cho mượn vượt số bản kể cả khi hai thủ thư cùng thao tác. Vì vậy toàn bộ việc kiểm tra và ghi phiếu được thực hiện trong một giao dịch, dòng sách và dòng độc giả được khóa bằng \texttt{SELECT \ldots{} FOR UPDATE} (Thuật toán~\ref{alg:muon-sach}).

\begin{algorithm}[H]
\caption{Lập phiếu mượn sách}
\label{alg:muon-sach}
\begin{algorithmic}[1]
\Require Mã sách $b$, mã độc giả $r$, số ngày mượn $d$
\Ensure Phiếu mượn mới hoặc thông báo lỗi
\State \textbf{BEGIN} giao dịch
\State Khóa dòng sách $b$; \textbf{if} không tồn tại \textbf{then} báo lỗi 404
\If{số phiếu đang mượn của $b \geq$ quantity}
    \State \textbf{ROLLBACK}; báo lỗi ``Sách đã được mượn hết''
\EndIf
\State Khóa dòng độc giả $r$, đếm số sách đang mượn $n$ và số sách quá hạn $q$
\If{thẻ hết hạn \textbf{or} $q > 0$ \textbf{or} $n \geq 5$ \textbf{or} $r$ đang mượn $b$}
    \State \textbf{ROLLBACK}; báo lỗi tương ứng
\EndIf
\State Thêm phiếu: due\_date $\gets$ hiện tại $+ d$ ngày
\State \textbf{COMMIT}; tăng metric \texttt{library\_loan\_operations\_total\{operation="borrow"\}}
\State \Return phiếu mượn
\end{algorithmic}
\end{algorithm}

Khi trả sách, tiền phạt được tính theo Công thức~\ref{eq:phat}:
\begin{equation}
\text{Phạt} = \max\left(0, \left\lceil \frac{t_{\text{trả}} - t_{\text{hạn}}}{1\ \text{ngày}} \right\rceil\right) \times 5000\ \text{(đồng)}
\label{eq:phat}
\end{equation}
